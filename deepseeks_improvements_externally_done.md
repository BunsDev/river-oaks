  const curve = new THREE.CatmullRomCurve3([
    new THREE.Vector3(1.4 * side, 0.35, 1.8),
    new THREE.Vector3(1.85 * side, 0.55, 0.8),
    new THREE.Vector3(1.95 * side, 0.5, -0.2),

  // profile - thin airfoil-like
  shape.moveTo(0, 0);
  shape.bezierCurveTo(0.02, 0.04, 0.06, 0.04, 0.08, 0);

    const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), mat);
    m.position.copy(pos);
    if (rot) m.rotation.copy(rot);
    envScene.add(m);
  };

  // Soft overhead key
  addPanel(0xffffff, 1.0, 12, 12,
    new THREE.Vector3(0, 8, 0), new THREE.Euler(-Math.PI / 2, 0, 0));

  // Cool fill from left
  addPanel(0xaaccff, 0.6, 1.2, 8,
    new THREE.Vector3(-7, 3, 0), new THREE.Euler(0, Math.PI / 2, 0));

  // Warm rim from right
  addPanel(0xffd9a0, 0.7, 1.2, 8,
    new THREE.Vector3(7, 3, 0), new THREE.Euler(0, -Math.PI / 2, 0));

  // Front kicker
  addPanel(0xffffff, 0.5, 6, 2,
    new THREE.Vector3(0, 1.5, 8), new THREE.Euler(0, Math.PI, 0));

  const pmrem = new THREE.PMREMGenerator(renderer);
  pmrem.compileEquirectangularShader();
  const tex = pmrem.fromScene(envScene, 0.03).texture;
  pmrem.dispose();
  envScene.traverse(o => {
    if (o.isMesh) { o.geometry.dispose(); o.material.dispose(); }
  });
  return tex;
}

function makeQuiltTexture(size = 256, color = 0xffffff) {
  const c = document.createElement('canvas');
  c.width = c.height = size;
  const ctx = c.getContext('2d');

  ctx.fillStyle = '#808080';
  ctx.fillRect(0, 0, size, size);

  const cells = 4;
  const s = size / cells;
  for (let y = 0; y < cells; y++) {
    for (let x = 0; x < cells; x++) {
      const cx = x * s + s / 2;
      const cy = y * s + s / 2;
      const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, s * 0.7);
      g.addColorStop(0.0, '#ffffff');
      g.addColorStop(0.45, '#c0c0c0');
      g.addColorStop(1.0, '#505050');
      ctx.fillStyle = g;
      ctx.fillRect(x * s, y * s, s, s);
    }
  }

  // Stitch grid
  ctx.strokeStyle = '#3a3a3a';
  ctx.lineWidth = Math.max(1, size / 128);
  for (let i = 0; i <= cells; i++) {
    const p = i * s;
    ctx.beginPath(); ctx.moveTo(p, 0); ctx.lineTo(p, size); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, p); ctx.lineTo(size, p); ctx.stroke();
  }

  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.anisotropy = 8;
  return tex;
}

function makeQuiltNormal(size = 256) {
  // Cheap normal map: derive from a radial bump pattern
  const c = document.createElement('canvas');
  c.width = c.height = size;
  const ctx = c.getContext('2d');
  const img = ctx.createImageData(size, size);
  const cells = 4;
  const s = size / cells;
  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const cx = (x % s) / s - 0.5;
      const cy = (y % s) / s - 0.5;
      const r = Math.sqrt(cx * cx + cy * cy);
      const n = Math.max(0, 1 - r * 2);
      const i = (y * size + x) * 4;
      img.data[i + 0] = 128 + cx * 60 * n;
      img.data[i + 1] = 128 + cy * 60 * n;
      img.data[i + 2] = 128 + n * 90;
      img.data[i + 3] = 255;
    }
  }
  ctx.putImageData(img, 0, 0);
  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  return tex;
}

function ribbonWing(side, M) {
  // Swept airfoil ribbon — the namesake.
  const curve = new THREE.CatmullRomCurve3([
    new THREE.Vector3(1.55 * side, 0.42,  1.85),
    new THREE.Vector3(1.94 * side, 0.60,  0.95),
    new THREE.Vector3(2.02 * side, 0.55, -0.15),
    new THREE.Vector3(1.86 * side, 0.34, -1.10),
    new THREE.Vector3(1.32 * side, 0.18, -1.92),
  ]);

  const shape = new THREE.Shape();
  shape.moveTo(0, 0);
  shape.bezierCurveTo(0.015, 0.07, 0.07, 0.07, 0.085, 0);
  shape.bezierCurveTo(0.07, -0.03, 0.015, -0.03, 0, 0);

  const geo = new THREE.ExtrudeGeometry(shape, {
    steps: 96,
    extrudePath: curve,
    bevelEnabled: true,
    bevelThickness: 0.008,
    bevelSize: 0.008,
    bevelSegments: 2,
  });
  geo.computeVertexNormals();

  const mesh = new THREE.Mesh(geo, M.lacquer);
  mesh.name = `ribbonWing${side > 0 ? 'R' : 'L'}`;

  // Accent edge — thin champagne strip along the outer edge
  const edgeCurve = new THREE.CatmullRomCurve3(
    curve.getPoints(64).map(p => p.clone().add(new THREE.Vector3(0.04 * side, 0.06, 0)))
  );
  const edge = new THREE.Mesh(
    new THREE.TubeGeometry(edgeCurve, 96, 0.008, 6, false),
    M.accent
  );

  const grp = new THREE.Group();
  grp.add(mesh, edge);
  return grp;
}

  const group = new THREE.Group();

  group.add(taillights(M));

  // Cache for animation
  group.userData = { materials: M, envMap };
  return group;
}

/* ═══════════════════════════════════════════════════════════
   SHOWCASE MODE
   Turntable + hover + thruster pulse. Call tick() each frame.
   ═══════════════════════════════════════════════════════════ */

export function showcaseMode(ribbonwing, opts = {}) {
  const { speed = 0.25, hover = 0.02, thrusterPulse = true } = opts;
  const baseY = ribbonwing.position.y;
  let t = 0;

  return {
    tick(dt) {
      t += dt;
      ribbonwing.rotation.y = t * speed;

      // Gentle hover
      ribbonwing.position.y = baseY + Math.sin(t * 1.4) * hover;

      if (thrusterPulse) {
        const pulse = 0.85 + Math.sin(t * 3.2) * 0.15;
        const M = ribbonwing.userData.materials;
        M.thrusterCore.emissiveIntensity = 8.0 * pulse;
        M.thrusterGlow.opacity = 0.85 * pulse;
        M.headlight.emissiveIntensity = 6.0 + Math.sin(t * 2.0) * 0.4;
      }
    },
  };
}
